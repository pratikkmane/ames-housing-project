Neighborhood

Code 
# Top 10 neighborhoods by count
top_neighborhoods <- ames %>%
  count(Neighborhood, sort = TRUE) %>%
  slice_head(n = 10) %>%
  pull(Neighborhood)

ames %>%
  filter(Neighborhood %in% top_neighborhoods) %>%
  group_by(Neighborhood) %>%
  summarise(
    Count = n(),
    Median_Price = median(SalePrice)
  ) %>%
  arrange(desc(Median_Price)) %>%
  ggplot(aes(x = reorder(Neighborhood, Median_Price), y = Median_Price)) +
  geom_col(fill = "#9b59b6", alpha = 0.8) +
  coord_flip() +
  scale_y_continuous(labels = dollar_format()) +
  labs(title = "Top 10 Neighborhoods by Median Price",
       x = "Neighborhood", y = "Median Sale Price") +
  theme_minimal()

Interpretation: Significant price variation by neighborhood. NridgHt and NoRidge are premium areas ($300K+), while MeadowV and BrDale are budget areas ($100K).
